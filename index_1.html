}); return Math.max.apply(null, ids) + 1; } function rowFromObj_(name, obj) { return SCHEMA[name].map(function(h){ var v = obj[h]; if (BOOL_COLS.indexOf(h) >= 0) return (v === true || String(v).toUpperCase() === 'TRUE');
